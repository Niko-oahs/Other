# -----------------------------------防火墙操作（防火墙设置-前置脚本中已安装防火墙-开启socks，vless，xui面板，door，四个端口）
打开防火墙
sudo systemctl start firewalld
设置防火墙开机启动
sudo systemctl enable firewalld

关闭端口
sudo firewall-cmd --zone=public --remove-port=xxxx/tcp --permanent
添加开放端口
sudo firewall-cmd --zone=public --add-port=xxxx/tcp --permanent
重新加载防火墙
sudo firewall-cmd --reload
查看防火墙开放端口
sudo firewall-cmd --list-all
查询防火墙状态
systemctl status firewalld


# ----------------------------------- 封禁脚本使用方法
# 查看状态
ban-manager status
# 查看封禁列表
ban-manager list
# 添加非标准SSH端口监控
ban-manager add-port 2222
# 添加白名单IP
ban-manager whitelist add 192.168.1.100
# 解封IP
ban-manager unban 1.2.3.4
